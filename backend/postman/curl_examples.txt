# Login (get JWT)
curl -X POST http://localhost:8080/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"username":"assistanttester","password":"Test@1234"}'

# Create requisition (replace TOKEN with returned JWT)
curl -X POST http://localhost:8080/api/requisitions \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer TOKEN" \
  -d '{
    "title":"New Laptops",
    "justification":"Replace old machines",
    "neededBy":"2026-08-15",
    "categoryId":1,
    "supplierId":1,
    "items":[{"description":"Dell Latitude","quantity":5,"unitPrice":1200.00}]
  }'

# Get my requisitions
curl -X GET http://localhost:8080/api/requisitions/my \
  -H "Authorization: Bearer TOKEN"

# Approve requisition (replace :id and TOKEN)
curl -X POST http://localhost:8080/api/requisitions/:id/actions \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer TOKEN" \
  -d '{"action":"APPROVE","remarks":"Approved via curl"}'
